function arena(t){x.fillStyle=grad(0,H,['#1a1a3a','#0a0a18']);x.fillRect(0,0,W,H);crowd(200,620,t,['#fff','#5b8cff','#c1121f','#ffd23f']);x.fillStyle='#2a5cff';x.fillRect(0,1000,W,H-1000);x.fillStyle='#1a3aa8';x.fillRect(0,1000,W,20);}
function ubars(cx){x.strokeStyle='#ddd';x.lineWidth=12;x.beginPath();x.moveTo(cx-260,1000);x.lineTo(cx-260,720);x.moveTo(cx+260,1000);x.lineTo(cx+260,720);x.moveTo(cx-140,1000);x.lineTo(cx-140,580);x.moveTo(cx+140,1000);x.lineTo(cx+140,580);x.stroke();x.strokeStyle='#f3e6c4';x.lineWidth=14;x.beginPath();x.moveTo(cx-280,720);x.lineTo(cx+280,720);x.moveTo(cx-160,580);x.lineTo(cx+160,580);x.stroke();}
function gym(cx,base,s,t,pose=0,rot=0,col='#e8e8f8'){x.save();x.translate(cx,base);x.rotate(rot);x.translate(-cx,-base);player(cx,base,s,col,col,pose);x.restore();}
function swing(cx,cy,t,s=0.8){x.save();x.translate(cx,cy);x.rotate(t*4.5);x.translate(0,340*s);x.rotate(Math.PI);player(0,0,s,'#e8e8f8','#e8e8f8',0);x.restore();}
function sboard(txt,cy,p,warn=0){x.save();x.globalAlpha=cl(p*2);x.fillStyle='#111';rr(190,cy-170,700,340,26);x.fill();x.strokeStyle='#555';x.lineWidth=10;rr(190,cy-170,700,340,26);x.stroke();x.fillStyle='#222';rr(230,cy-120,620,240,14);x.fill();x.fillStyle=warn?'#ff4d4d':'#ffd23f';x.font=`900 190px monospace`;x.textAlign='center';x.textBaseline='middle';x.shadowColor=x.fillStyle;x.shadowBlur=30;x.fillText(txt,540,cy+10);x.restore();}
