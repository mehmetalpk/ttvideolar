function mirror(cx,cy,s,crack){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#6a5a3a';x.beginPath();x.ellipse(0,0,230,320,0,0,7);x.fill();const g=x.createLinearGradient(-200,-300,200,300);g.addColorStop(0,'#9fb4c8');g.addColorStop(1,'#3a4a5a');x.fillStyle=g;x.beginPath();x.ellipse(0,0,200,290,0,0,7);x.fill();x.fillStyle='rgba(0,0,0,0.75)';x.beginPath();x.arc(0,-60,60,0,7);x.fill();rr(-80,10,160,200,50);x.fill();if(crack>0){x.strokeStyle='#fff';x.lineWidth=5;for(let i=0;i<7;i++){x.beginPath();x.moveTo(40,-120);x.lineTo(40+Math.cos(i*0.9)*220*crack,-120+Math.sin(i*0.9)*220*crack);x.stroke();}}x.restore();}
function sign(p){x.save();x.strokeStyle='#ffd23f';x.lineWidth=14;x.lineCap='round';x.beginPath();const n=Math.floor(60*cl(p));for(let i=0;i<=n;i++){const u=i/60;const px=220+u*640,py=760+Math.sin(u*18)*60*Math.sin(u*3.1)-u*40;i?x.lineTo(px,py):x.moveTo(px,py);}x.stroke();x.restore();}
function pushup(cx,base,s,t){const d=Math.abs(Math.sin(t*3))*30;x.save();x.translate(cx,base);x.scale(s,s);x.strokeStyle='#ffd23f';x.lineCap='round';x.lineWidth=26;x.beginPath();x.moveTo(-220,0);x.lineTo(80,-60+d*0.5);x.stroke();x.lineWidth=20;x.beginPath();x.moveTo(60,-60+d*0.5);x.lineTo(70,0);x.stroke();x.fillStyle='#ffd23f';x.beginPath();x.arc(140,-80+d*0.6,34,0,7);x.fill();x.restore();}
