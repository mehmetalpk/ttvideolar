function silverPile(cx,base,n,s){let k=0;for(let r=0;k<n;r++){const m=Math.max(1,6-r);for(let i=0;i<m&&k<n;i++,k++)sbar(cx+(i-(m-1)/2)*200*s,base-r*70*s,s);}}
function derrick(cx,base,s,t){x.save();x.translate(cx,base);x.scale(s,s);x.strokeStyle='#2a2a2a';x.lineWidth=12;x.beginPath();x.moveTo(-120,0);x.lineTo(0,-460);x.lineTo(120,0);for(let i=1;i<6;i++){const y=-i*76,w=120-i*20;x.moveTo(-w,y);x.lineTo(w,y);}x.stroke();x.restore();}
function cowboy(cx,base,s,col){sil(cx,base,s,col);x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle=col;x.beginPath();x.ellipse(0,-205,90,18,0,0,7);x.fill();rr(-45,-260,90,60,20);x.fill();x.restore();}
