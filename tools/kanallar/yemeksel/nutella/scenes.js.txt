function nut(cx,cy,s,rot=0){x.save();x.translate(cx,cy);x.rotate(rot);x.scale(s,s);x.fillStyle='#a8662a';x.beginPath();x.ellipse(0,10,46,50,0,0,7);x.fill();x.fillStyle='#d9a86a';x.beginPath();x.ellipse(0,-34,40,18,0,0,7);x.fill();x.fillStyle='rgba(255,255,255,0.25)';x.beginPath();x.ellipse(-16,4,10,22,0.3,0,7);x.fill();x.restore();}
function nutRain(t,n=16){const r=rng(9);for(let i=0;i<n;i++){const px=r()*W,sp=250+r()*300,py=((r()*H+t*sp)%(H+200))-100;nut(px,py,0.9+r()*0.6,t*2+i);}}
function cocoa(cx,cy,s){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#8a4a1a';x.beginPath();x.ellipse(0,0,70,130,0,0,7);x.fill();x.strokeStyle='#5a2a0a';x.lineWidth=6;for(let i=-2;i<=2;i++){x.beginPath();x.moveTo(i*22,-120);x.quadraticCurveTo(i*30,0,i*22,120);x.stroke();}x.restore();}
function loaf(cx,cy,s,cut){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#5a3218';rr(-240,-90,480,180,30);x.fill();x.fillStyle='#7a4a24';rr(-240,-90,480,40,20);x.fill();if(cut){x.strokeStyle='#3a1e0c';x.lineWidth=6;for(let i=1;i<6;i++){x.beginPath();x.moveTo(-240+i*80,-90);x.lineTo(-240+i*80,90);x.stroke();}}x.restore();}
function breadSl(cx,cy,s,spread){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#e8c48a';x.beginPath();x.moveTo(-150,120);x.lineTo(-150,-60);x.quadraticCurveTo(-150,-160,0,-160);x.quadraticCurveTo(150,-160,150,-60);x.lineTo(150,120);x.closePath();x.fill();x.fillStyle='#f7e6c4';x.beginPath();x.moveTo(-125,100);x.lineTo(-125,-55);x.quadraticCurveTo(-125,-135,0,-135);x.quadraticCurveTo(125,-135,125,-55);x.lineTo(125,100);x.closePath();x.fill();if(spread>0){x.fillStyle='#5a3218';x.beginPath();x.ellipse(0,-10,115*spread,100*spread,0,0,7);x.fill();}x.restore();}
function jarN(cx,cy,s,txt){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='rgba(255,255,255,0.3)';rr(-130,-170,260,340,40);x.fill();x.fillStyle='#4a2810';rr(-118,-120,236,280,34);x.fill();x.fillStyle='#f3efe6';x.fillRect(-120,-215,240,60);x.fillStyle='#fff8e6';rr(-110,-40,220,110,16);x.fill();x.fillStyle='#3a2414';x.font=`900 56px ${F}`;x.textAlign='center';x.textBaseline='middle';x.fillText(txt,0,15);x.restore();}
