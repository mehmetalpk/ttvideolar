function cone(cx,cy,s,scoops=1,rot=0){x.save();x.translate(cx,cy);x.rotate(rot);x.scale(s,s);x.fillStyle='#d99a4a';x.beginPath();x.moveTo(-80,-40);x.lineTo(80,-40);x.lineTo(0,200);x.closePath();x.fill();x.save();x.clip();x.strokeStyle='#a86a2a';x.lineWidth=5;for(let i=-6;i<=6;i++){x.beginPath();x.moveTo(i*30-100,-40);x.lineTo(i*30+100,200);x.stroke();x.beginPath();x.moveTo(i*30+100,-40);x.lineTo(i*30-100,200);x.stroke();}x.restore();const cols=['#f4e0f0','#7a4a2a','#ffd0a0'];for(let k=0;k<scoops;k++){x.fillStyle=cols[k%3];x.beginPath();x.arc(0,-60-k*80,78,Math.PI*0.95,Math.PI*2.05);x.fill();x.beginPath();x.ellipse(0,-50-k*80,84,22,0,0,7);x.fill();}x.restore();}
function dish(cx,cy,s,empty=0){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#e8eef4';x.beginPath();x.moveTo(-110,-20);x.quadraticCurveTo(0,120,110,-20);x.fill();x.fillRect(-14,40,28,60);x.fillRect(-60,96,120,16);if(!empty){x.fillStyle='#f4e0f0';x.beginPath();x.arc(0,-20,70,Math.PI,0);x.fill();}x.restore();}
