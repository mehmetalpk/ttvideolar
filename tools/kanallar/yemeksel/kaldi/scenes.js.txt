function goat(cx,base,s,t,jump=0,col='#f4f1ea'){const j=jump?Math.abs(Math.sin(t*7+cx))*90:0;x.save();x.translate(cx,base-j*s);x.scale(s,s);x.rotate(jump?Math.sin(t*7+cx)*0.15:0);x.fillStyle=col;x.beginPath();x.ellipse(0,-120,110,60,0,0,7);x.fill();x.strokeStyle=col;x.lineWidth=18;x.lineCap='round';for(const lx of [-70,-30,40,80]){x.beginPath();x.moveTo(lx,-80);x.lineTo(lx+(jump?Math.sin(t*14+lx)*15:0),0);x.stroke();}x.beginPath();x.ellipse(120,-190,40,30,-0.5,0,7);x.fill();x.beginPath();x.moveTo(95,-210);x.lineTo(80,-260);x.stroke();x.fillStyle='#2a1a10';x.beginPath();x.arc(132,-198,6,0,7);x.fill();x.fillStyle=col;x.beginPath();x.moveTo(130,-165);x.lineTo(125,-130);x.lineTo(145,-160);x.fill();x.restore();}
function bush(cx,cy,s){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#2f6a2a';for(let i=0;i<6;i++){x.beginPath();x.arc(-80+i*32,-40-(i%2)*40,70,0,7);x.fill();}x.fillStyle='#d8202a';for(let i=0;i<14;i++){x.beginPath();x.arc(-120+(i*53)%240,-90+(i*37)%110,13,0,7);x.fill();}x.restore();}
function hills(t,night=0){x.fillStyle=grad(0,1000,night?['#0a1030','#2a2a5a']:['#f4c07a','#e08a4a']);x.fillRect(0,0,W,H);if(night){x.fillStyle='#f4f1ea';x.beginPath();x.arc(820,300,60,0,7);x.fill();x.fillStyle='#fff';for(let i=0;i<30;i++)x.fillRect((i*197)%W,(i*113)%600,4,4);}x.fillStyle=night?'#1a2a1a':'#6a8a3a';x.beginPath();x.moveTo(0,1000);for(let i=0;i<=10;i++)x.lineTo(i*108,920-Math.sin(i*1.3)*60);x.lineTo(W,H);x.lineTo(0,H);x.fill();}
function monastery(cx,base,s){x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle='#c8a878';x.fillRect(-260,-300,520,300);x.fillStyle='#8a5a3a';x.beginPath();x.moveTo(-300,-300);x.lineTo(0,-470);x.lineTo(300,-300);x.fill();x.fillStyle='#4a2e1a';rr(-60,-180,120,180,50);x.fill();x.restore();}
