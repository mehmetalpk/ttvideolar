function badge(cx,cy,txt,col,p){if(p<=0)return;x.save();x.translate(cx,cy);const s=back(cl(p));x.scale(s,s);x.font=`900 56px ${F}`;const w=Math.max(380,x.measureText(txt).width+80);x.fillStyle=col;rr(-w/2,-60,w,120,60);x.fill();x.fillStyle='#fff';x.textAlign='center';x.textBaseline='middle';x.fillText(txt,0,4);x.restore();}
function stage(t){x.fillStyle='#0a0a10';x.fillRect(0,0,W,H);x.fillStyle='rgba(255,240,200,0.2)';x.beginPath();x.moveTo(440,0);x.lineTo(640,0);x.lineTo(820,1300);x.lineTo(260,1300);x.fill();x.fillStyle='#3a1c1c';x.fillRect(0,1300,W,H);}
function coins(t,n=18,seed=3){const r=rng(seed);for(let i=0;i<n;i++){const xx=r()*W,sp=300+r()*400,yy=((r()*H+t*sp)%(H+200))-100;x.save();x.translate(xx,yy);x.scale(Math.max(0.05,Math.abs(Math.cos(t*4+i))),1);x.fillStyle='#e0b640';x.beginPath();x.arc(0,0,30,0,7);x.fill();x.strokeStyle='#a07a20';x.lineWidth=5;x.stroke();x.restore();}}
function stampX(cx,cy,s,p){if(p<=0)return;const sc=1+(1-back(p))*0.8;x.save();x.translate(cx,cy);x.scale(s*sc,s*sc);x.rotate(-0.15);x.globalAlpha=cl(p*2);x.strokeStyle='#c1121f';x.lineWidth=26;x.lineCap='round';x.beginPath();x.moveTo(-80,-80);x.lineTo(80,80);x.moveTo(80,-80);x.lineTo(-80,80);x.stroke();x.restore();}
function fire(cx,base,w,h,t,a=1){const n=7;for(let i=0;i<n;i++){const k=i/(n-1)-0.5,fh=h*(1-Math.abs(k)*1.2)*(0.8+0.2*Math.sin(t*9+i*2)),fw=w/n*1.8,px=cx+k*w;if(fh<=0)continue;const g=x.createLinearGradient(0,base-fh,0,base);g.addColorStop(0,'rgba(255,230,120,0)');g.addColorStop(0.35,`rgba(255,170,40,${0.9*a})`);g.addColorStop(1,`rgba(210,50,20,${0.95*a})`);x.fillStyle=g;x.beginPath();x.moveTo(px-fw/2,base);x.quadraticCurveTo(px-fw*0.6,base-fh*0.5,px+Math.sin(t*5+i)*10,base-fh);x.quadraticCurveTo(px+fw*0.6,base-fh*0.5,px+fw/2,base);x.fill();}}
function mapArrow(ax,ay,bx,by,p,col='#ff3b3b'){if(p<=0)return;const q=cl(p);x.save();x.strokeStyle=col;x.lineWidth=12;x.setLineDash([26,18]);x.beginPath();x.moveTo(ax,ay);const mx=(ax+bx)/2,my=Math.min(ay,by)-150;const N=30;for(let i=1;i<=N*q;i++){const u=i/N;const px=(1-u)*(1-u)*ax+2*(1-u)*u*mx+u*u*bx,py=(1-u)*(1-u)*ay+2*(1-u)*u*my+u*u*by;x.lineTo(px,py);}x.stroke();x.setLineDash([]);x.fillStyle=col;x.beginPath();x.arc(ax,ay,16,0,7);x.fill();if(q>=1){x.beginPath();x.arc(bx,by,22,0,7);x.fill();}x.restore();}
function counter(v0,v1,p,cx,cy,sz,col,suf=''){const v=Math.round(v0+(v1-v0)*eOut(cl(p)));bigText(v.toLocaleString('tr-TR')+suf,cx,cy,sz,col,1);}
