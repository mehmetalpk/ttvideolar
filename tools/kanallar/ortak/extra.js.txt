 x.fillStyle='#4f3f37';x.beginPath();x.moveTo(-500,500);x.lineTo(-250,-250);x.lineTo(150,-330);x.lineTo(500,500);x.fill();
 x.fillStyle='#0a0705';x.beginPath();x.moveTo(-260,500);x.lineTo(-260,40);x.quadraticCurveTo(0,-260,260,40);x.lineTo(260,500);x.fill();
 if(glow){const g=x.createRadialGradient(0,300,10,0,300,320);g.addColorStop(0,`rgba(255,215,130,${glow})`);g.addColorStop(1,'rgba(255,215,130,0)');x.fillStyle=g;x.fillRect(-300,-50,600,560);}x.restore();}

function coin(cx,cy,r,t,txt){const sx=Math.abs(Math.cos(t*2));x.save();x.translate(cx,cy);x.scale(Math.max(0.08,sx),1);const g=x.createLinearGradient(-r,-r,r,r);g.addColorStop(0,'#f4f4f4');g.addColorStop(0.5,'#a9adb3');g.addColorStop(1,'#e6e6e6');x.fillStyle=g;x.beginPath();x.arc(0,0,r,0,7);x.fill();x.strokeStyle='#6c7077';x.lineWidth=r*0.06;x.beginPath();x.arc(0,0,r*0.82,0,7);x.stroke();star8(0,0,r*0.45,'#6c7077',r*0.04);x.restore();}
function sun(cx,cy,r){const g=x.createRadialGradient(cx,cy,r*0.3,cx,cy,r*3);g.addColorStop(0,'rgba(255,220,120,0.9)');g.addColorStop(1,'rgba(255,180,80,0)');x.fillStyle=g;x.fillRect(0,0,W,H);x.fillStyle='#ffd77a';x.beginPath();x.arc(cx,cy,r,0,7);x.fill();}
function city(base,col,dome){const r=rng(11);let px=-40;x.fillStyle=col;while(px<W+40){const w=120+r()*110,h=160+r()*260;x.fillRect(px,base-h,w,h);if(dome&&r()<0.5){x.beginPath();x.arc(px+w/2,base-h,w/2.4,Math.PI,0);x.fill();}px+=w+6;}}
function waves(y,t,col,amp=30,len=180){x.fillStyle=col;x.beginPath();x.moveTo(0,H);for(let px=0;px<=W;px+=20)x.lineTo(px,y+Math.sin(px/len*6.28+t*3)*amp+Math.sin(px/70+t*5)*amp*0.25);x.lineTo(W,H);x.fill();}
