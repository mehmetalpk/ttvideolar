 for(let i=0;i<7;i++){const a=-Math.PI/2+(i-3)*0.32;const L=60+wing*170;x.save();x.rotate(a+Math.PI/2);x.fillStyle=i%2?PAL.red:PAL.gold;x.beginPath();x.moveTo(0,-40);x.lineTo(-18,-40-L);x.lineTo(18,-40-L);x.closePath();if(wing>0)x.fill();x.restore();}
 x.fillStyle='#7a5230';rr(-55,-150,110,270,20);x.fill();x.fillStyle=PAL.gold;x.fillRect(-55,-60,110,14);x.fillRect(-55,40,110,14);x.fillStyle=PAL.red;x.beginPath();x.moveTo(-55,-150);x.lineTo(0,-250);x.lineTo(55,-150);x.fill();x.fillStyle='#2b1d12';x.beginPath();x.arc(0,-100,22,0,7);x.fill();x.restore();}
function smoke(cx,cy,t,n=10){const r=rng(5);for(let i=0;i<n;i++){const k=((t*0.6+r())%1);x.fillStyle=`rgba(220,220,230,${0.45*(1-k)})`;x.beginPath();x.arc(cx+(r()-0.5)*160*k,cy+k*500,40+k*90,0,7);x.fill();}}
function fireworks(t,seed=2){const r=rng(seed);for(let i=0;i<5;i++){const k=((t*0.7+i*0.2)%1),cx=150+r()*780,cy=300+r()*500,col=['#ffd23f','#ff5a5a','#7fd3ff','#c9ff7f'][i%4];x.fillStyle=col;x.globalAlpha=1-k;for(let j=0;j<14;j++){const a=j/14*6.28;x.beginPath();x.arc(cx+Math.cos(a)*k*170,cy+Math.sin(a)*k*170,6,0,7);x.fill();}x.globalAlpha=1;}}
