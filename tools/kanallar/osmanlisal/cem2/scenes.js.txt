function tSil(cx,base,s,col='#1a0f08',tur='#f3e6c4'){s*=1.7;sil(cx,base,s,col);x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle=tur;x.beginPath();x.ellipse(0,-182,38,26,0,0,7);x.fill();x.beginPath();x.ellipse(0,-204,24,18,0,0,7);x.fill();x.fillStyle=PAL.red;x.fillRect(-4,-222,8,14);x.restore();}
function throneO(cx,base,s,glow=0){x.save();x.translate(cx,base);x.scale(s,s);if(glow)spot(0,-200,380,0.35*glow,'255,215,120');x.fillStyle='#8a5a1a';rr(-150,-420,300,420,30);x.fill();x.fillStyle=PAL.red;rr(-110,-370,220,230,24);x.fill();x.fillStyle=PAL.gold;x.fillRect(-190,-150,380,50);rr(-200,-30,400,30,8);x.fill();x.restore();}
function fortress(cx,base,s,col='#5a5048'){x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle=col;x.fillRect(-320,-260,640,260);for(let i=0;i<9;i++)x.fillRect(-320+i*76,-300,44,44);x.fillRect(-380,-420,120,420);x.fillRect(260,-420,120,420);for(let k of [-380,260])for(let i=0;i<3;i++)x.fillRect(k+i*44,-460,30,40);x.fillStyle='#1a1410';x.beginPath();x.moveTo(-60,0);x.lineTo(-60,-120);x.quadraticCurveTo(0,-180,60,-120);x.lineTo(60,0);x.fill();x.fillStyle='rgba(255,210,120,0.6)';x.fillRect(-340,-360,30,40);x.fillRect(300,-360,30,40);x.restore();}
function armyRow(base,n,s,t,col,dir=0){for(let i=0;i<n;i++)janissary(80+i*(W-160)/(n-1)+Math.sin(t*4+i)*3+dir*t*40,base,s,col);}
function mapO(t){paper(false);x.fillStyle='rgba(120,100,60,0.3)';[[300,640,260,170],[700,620,240,150],[560,900,300,140],[820,1000,180,200]].forEach(([a,b,c,d])=>{x.beginPath();x.ellipse(a,b,c,d,0.2,0,7);x.fill();});x.fillStyle='rgba(60,110,150,0.25)';x.fillRect(0,760,W,60);}
function dot(cx,cy,lab,col=PAL.red,p=1){if(p<=0)return;x.fillStyle=col;x.beginPath();x.arc(cx,cy,18,0,7);x.fill();bigText(lab,cx,cy-60,62,'#fff',eOut(cl(p)));}
function seaN(t){x.fillStyle=grad(0,H,['#0a1428','#1a3050']);x.fillRect(0,0,W,H);crescent(820,420,60,'#f3e6c4');x.fillStyle='#16324f';x.fillRect(0,1060,W,H-1060);waves(1066,t,'#24486a',8);}
function chains(cx,cy,t){x.strokeStyle='#888';x.lineWidth=10;for(let i=0;i<7;i++){x.beginPath();x.ellipse(cx-150+i*50,cy+Math.sin(t*3+i)*6,26,16,0,0,7);x.stroke();}}
function goldStack(cx,base,n){for(let i=0;i<n;i++){x.fillStyle=i%2?'#e0b640':'#c99a2a';x.beginPath();x.ellipse(cx,base-i*22,90,26,0,0,7);x.fill();x.strokeStyle='#8a6a12';x.lineWidth=3;x.stroke();}}
function knight(cx,base,s,col='#2a2d36'){s*=1.7;x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle=col;x.beginPath();x.moveTo(-42,0);x.lineTo(-36,-120);x.quadraticCurveTo(0,-150,36,-120);x.lineTo(42,0);x.fill();x.fillStyle='#8a8f99';rr(-28,-196,56,66,14);x.fill();x.fillStyle='#111';x.fillRect(-20,-172,40,6);x.fillStyle='#c1121f';x.fillRect(-30,-110,60,70);x.fillStyle='#f3f3f3';x.fillRect(-6,-104,12,58);x.fillRect(-24,-82,48,12);x.restore();}
