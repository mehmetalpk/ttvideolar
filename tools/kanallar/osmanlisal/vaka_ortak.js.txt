function kazan(cx,cy,s,tilt=0,t=0){x.save();x.translate(cx,cy);x.rotate(tilt);x.scale(s,s);x.fillStyle='#8a4a1e';x.beginPath();x.moveTo(-170,-60);x.quadraticCurveTo(-180,120,0,130);x.quadraticCurveTo(180,120,170,-60);x.closePath();x.fill();x.fillStyle='#b8692a';x.beginPath();x.ellipse(0,-60,175,36,0,0,7);x.fill();x.fillStyle='#3a1e0a';x.beginPath();x.ellipse(0,-60,150,26,0,0,7);x.fill();x.strokeStyle='#5a2e10';x.lineWidth=14;x.beginPath();x.arc(-175,-20,34,Math.PI*0.5,Math.PI*1.5);x.stroke();x.beginPath();x.arc(175,-20,34,-Math.PI*0.5,Math.PI*0.5);x.stroke();x.fillStyle='rgba(255,220,160,0.25)';x.beginPath();x.ellipse(-80,30,30,60,0.3,0,7);x.fill();x.restore();}
function throne(cx,base,s,shake=0,t=0){x.save();x.translate(cx+Math.sin(t*40)*shake*10,base);x.rotate(Math.sin(t*30)*shake*0.03);x.scale(s,s);x.fillStyle='#7a1d1d';rr(-150,-420,300,300,30);x.fill();x.fillStyle=PAL.gold;rr(-190,-140,380,60,12);x.fill();x.fillRect(-170,-80,30,80);x.fillRect(140,-80,30,80);x.fillStyle='#c9a24a';x.beginPath();x.arc(0,-430,40,0,7);x.fill();x.restore();}
function barracks(cx,base,s,col='#5a4a3a'){x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle=col;x.fillRect(-420,-260,840,260);x.beginPath();x.moveTo(-440,-260);x.lineTo(0,-340);x.lineTo(440,-260);x.fill();x.fillStyle='rgba(20,12,6,0.7)';for(let i=0;i<7;i++){rr(-380+i*112,-200,60,90,28);x.fill();}x.fillStyle='#2b1d12';rr(-60,-130,120,130,50);x.fill();x.restore();}
function cannon(cx,base,s,t,fire=0){x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle='#3a2414';x.beginPath();x.arc(-40,-40,46,0,7);x.fill();x.fillStyle='#1a1a1a';x.save();x.translate(0,-70);x.rotate(-0.25);rr(-60,-30,260,60,26);x.fill();if(fire>0){x.fillStyle=`rgba(255,200,80,${fire})`;x.beginPath();x.arc(230,0,70*fire+20,0,7);x.fill();x.fillStyle=`rgba(255,120,40,${fire})`;x.beginPath();x.arc(230,0,40*fire+10,0,7);x.fill();}x.restore();x.restore();}
function sancak(cx,cy,s,t){x.save();x.translate(cx,cy);x.scale(s,s);x.fillStyle='#2b1d12';x.fillRect(-6,-40,12,520);x.fillStyle=PAL.gold;x.beginPath();x.moveTo(-20,-40);x.lineTo(0,-90);x.lineTo(20,-40);x.fill();x.fillStyle='#1d5c3a';x.beginPath();x.moveTo(6,-30);for(let i=0;i<=10;i++)x.lineTo(6+i*30,-30+Math.sin(t*5+i*0.6)*10);for(let i=10;i>=0;i--)x.lineTo(6+i*30,190+Math.sin(t*5+i*0.6)*10);x.fill();x.strokeStyle=PAL.gold;x.lineWidth=6;x.strokeRect(30,0,250,160);x.restore();}
function smokeP(cx,cy,t,n=6){for(let i=0;i<n;i++){const k=((t*0.5+i/n)%1);x.fillStyle=`rgba(60,55,50,${0.5*(1-k)})`;x.beginPath();x.arc(cx+Math.sin(i*2)*60,cy-k*500,50+k*120,0,7);x.fill();}}
