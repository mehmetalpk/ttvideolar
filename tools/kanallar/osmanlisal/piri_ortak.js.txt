function parch(cx,cy,w,h,rot=0,p=1){x.save();x.translate(cx,cy);x.rotate(rot);x.scale(p,p);x.fillStyle='#d9b98a';x.beginPath();x.moveTo(-w/2,-h/2+20);x.quadraticCurveTo(-w/4,-h/2-10,0,-h/2+10);x.quadraticCurveTo(w/4,-h/2+30,w/2,-h/2);x.lineTo(w/2-20,h/2);x.quadraticCurveTo(0,h/2+20,-w/2+10,h/2-10);x.closePath();x.fill();x.strokeStyle='#8a6a3a';x.lineWidth=6;x.stroke();x.restore();}

 x.beginPath();x.moveTo(140,-380);x.quadraticCurveTo(100,-200,180,-60);x.quadraticCurveTo(220,80,150,240*k+(-60)*(1-k));x.lineTo(280,240*k);x.lineTo(280,-380);x.fill();
 x.beginPath();x.moveTo(-280,-300);x.quadraticCurveTo(-180,-260,-200,-120);x.quadraticCurveTo(-120,20,-180,180*k);x.quadraticCurveTo(-120,300,-260,380*k);x.lineTo(-280,380);x.fill();
 x.fillStyle='#c98a4a';for(let i=0;i<5;i++){x.beginPath();x.arc(-40+i*30,-120+Math.sin(i)*60,8,0,7);x.fill();}x.restore();
 if(half){x.fillStyle='rgba(20,12,6,0.85)';x.fillRect(80,-400,240,800);}

function captain(cx,base,s,col='#1a0f08'){s*=1.7;sil(cx,base,s,col);x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle='#f3e6c4';x.beginPath();x.ellipse(0,-180,40,24,0,0,7);x.fill();x.fillStyle=col;x.beginPath();x.moveTo(-20,-140);x.quadraticCurveTo(0,-90,20,-140);x.fill();x.restore();}
function compass(cx,cy,r,t){x.save();x.translate(cx,cy);x.rotate(Math.sin(t*2)*0.3);x.strokeStyle='#8a6a3a';x.lineWidth=6;x.beginPath();x.arc(0,0,r,0,7);x.stroke();x.fillStyle=PAL.red;x.beginPath();x.moveTo(0,-r*0.85);x.lineTo(r*0.15,0);x.lineTo(0,r*0.2);x.lineTo(-r*0.15,0);x.fill();x.restore();}
function galley(cx,cy,s,t,col='#3a2414',flag=PAL.red,row=1){x.save();x.translate(cx,cy);x.scale(s,s);if(row){x.strokeStyle='#2b1d12';x.lineWidth=5;for(let i=0;i<9;i++){const a=Math.sin(t*5+i*0.3)*0.4*row;x.beginPath();x.moveTo(-200+i*50,10);x.lineTo(-200+i*50-60*Math.sin(a+0.6),90);x.stroke();}}x.fillStyle=col;x.beginPath();x.moveTo(-280,-10);x.lineTo(300,-10);x.lineTo(340,-40);x.lineTo(240,40);x.lineTo(-230,40);x.closePath();x.fill();x.fillRect(-6,-260,12,250);x.fillStyle='#efe2c2';x.beginPath();x.moveTo(0,-250);x.lineTo(150,-40);x.lineTo(-10,-60);x.fill();x.fillStyle=flag;x.beginPath();x.moveTo(6,-260);x.lineTo(70,-245+Math.sin(t*6)*5);x.lineTo(6,-225);x.fill();x.restore();}
