function heightBar(cx,top,base,label,p){if(p<=0)return;x.save();x.globalAlpha=cl(p*2);x.strokeStyle='#ffd23f';x.lineWidth=8;const y=base-(base-top)*eOut(p);x.beginPath();x.moveTo(cx,base);x.lineTo(cx,y);x.stroke();x.beginPath();x.moveTo(cx-30,y);x.lineTo(cx+30,y);x.moveTo(cx-30,base);x.lineTo(cx+30,base);x.stroke();x.restore();bigText(label,cx,(top+base)/2,64,'#ffd23f',p,-Math.PI/2);}
function vault(t,lit){x.fillStyle=grad(0,H,['#3a2a1e','#120c08']);x.fillRect(0,0,W,H);x.fillStyle='#5a4632';x.beginPath();x.moveTo(0,700);x.quadraticCurveTo(540,80,W,700);x.lineTo(W,0);x.lineTo(0,0);x.fill();if(lit>0){x.save();x.beginPath();x.moveTo(0,700);x.quadraticCurveTo(540,80,W,700);x.lineTo(W,0);x.lineTo(0,0);x.clip();const cols=['#c98a5a','#6a8ab8','#d9b36a','#8aa86a','#b86a5a','#e0c89a'];for(let i=0;i<12;i++){const c=i%4,r=Math.floor(i/4);if(i/12>lit)continue;x.fillStyle=cols[i%6];rr(60+c*250,40+r*200,220,170,14);x.fill();x.fillStyle='rgba(0,0,0,0.25)';x.beginPath();x.arc(170+c*250,130+r*200,40,0,7);x.fill();}x.restore();}x.fillStyle='#2a1e14';for(let i=0;i<4;i++)x.fillRect(80+i*290,700,50,1300);}
function pope(cx,base,s){sil(cx,base,s,'#f2efe6');x.save();x.translate(cx,base-212*s);x.fillStyle='#f2efe6';x.beginPath();x.moveTo(-40*s,10*s);x.quadraticCurveTo(0,-110*s,40*s,10*s);x.fill();x.fillStyle='#ffd23f';x.fillRect(-4*s,-80*s,8*s,60*s);x.restore();}
function painter(cx,base,s,t,col='#1a1208'){x.save();x.translate(cx,base);x.scale(s,s);x.strokeStyle=col;x.fillStyle=col;x.lineCap='round';x.lineWidth=26;x.beginPath();x.moveTo(-20,0);x.lineTo(0,-150);x.lineTo(20,0);x.stroke();x.lineWidth=34;x.beginPath();x.moveTo(0,-150);x.lineTo(-20,-280);x.stroke();x.beginPath();x.arc(-40,-320,30,0,7);x.fill();x.lineWidth=16;const a=Math.sin(t*6)*0.3;x.beginPath();x.moveTo(-14,-260);x.lineTo(30+a*40,-360);x.lineTo(50+a*60,-430);x.stroke();x.strokeStyle='#c98a3a';x.lineWidth=8;x.beginPath();x.moveTo(50+a*60,-430);x.lineTo(60+a*60,-480);x.stroke();x.restore();}
function drips(cx,cy,t){x.fillStyle='#6a8ab8';for(let i=0;i<6;i++){const y=cy+((t*300+i*80)%400);x.beginPath();x.ellipse(cx-60+i*24,y,6,12,0,0,7);x.fill();}}
function bow(cx,cy,p){x.save();x.strokeStyle='#ffd23f';x.lineWidth=14;x.beginPath();x.arc(cx,cy+300,320,-Math.PI/2-1.0*p,-Math.PI/2+1.0*p);x.stroke();x.restore();}
