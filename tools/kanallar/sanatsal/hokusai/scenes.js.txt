function paper(){x.fillStyle=grad(0,H,['#efe2c4','#d8c49a']);x.fillRect(0,0,W,H);}
function fuji(cx,base,s){x.save();x.translate(cx,base);x.scale(s,s);x.fillStyle='#3a4a6a';x.beginPath();x.moveTo(-200,0);x.lineTo(-40,-200);x.lineTo(40,-200);x.lineTo(200,0);x.fill();x.fillStyle='#fff';x.beginPath();x.moveTo(-40,-200);x.lineTo(40,-200);x.lineTo(70,-160);x.lineTo(20,-170);x.lineTo(0,-150);x.lineTo(-30,-170);x.lineTo(-70,-160);x.fill();x.restore();}
function bigWave(cx,cy,s,t,p=1){x.save();x.translate(cx,cy);x.scale(s,s);x.globalAlpha*=cl(p*2);const o=Math.sin(t*2)*10;x.fillStyle='#1f3a6a';x.beginPath();x.moveTo(-500,300);x.bezierCurveTo(-450,-100,-150,-420,150+o,-380);x.bezierCurveTo(330,-350,380,-220,280,-160);x.bezierCurveTo(200,-260,60,-260,0,-120);x.bezierCurveTo(-60,40,100,200,500,300);x.fill();x.fillStyle='#fff';for(let i=0;i<9;i++){const a=i/8;const px=150-a*120+o,py=-380+a*40;x.beginPath();x.arc(px+Math.cos(i)*20,py,22-i,0,7);x.fill();}x.strokeStyle='rgba(255,255,255,0.6)';x.lineWidth=8;for(let i=0;i<4;i++){x.beginPath();x.moveTo(-420+i*60,280);x.bezierCurveTo(-380+i*60,-60,-120+i*40,-340,120,-340+i*30);x.stroke();}x.restore();}
function brush(cx,cy,t){x.save();x.translate(cx+Math.sin(t*6)*20,cy);x.rotate(0.5);x.fillStyle='#7a5a32';x.fillRect(-8,-200,16,180);x.fillStyle='#111';x.beginPath();x.moveTo(-12,-20);x.lineTo(12,-20);x.lineTo(0,30);x.fill();x.restore();}
