function eiffel(cx,base,s,p=1,col='#5a3a22'){x.save();x.translate(cx,base);x.scale(s,s);x.beginPath();x.rect(-400,-1000*p,800,1000*p+10);x.clip();x.strokeStyle=col;x.fillStyle=col;x.lineWidth=14;x.beginPath();x.moveTo(-260,0);x.quadraticCurveTo(-120,-300,-40,-700);x.lineTo(-14,-900);x.lineTo(14,-900);x.lineTo(40,-700);x.quadraticCurveTo(120,-300,260,0);x.stroke();x.beginPath();x.moveTo(-180,0);x.quadraticCurveTo(0,-200,180,0);x.stroke();x.fillRect(-170,-300,340,22);x.fillRect(-80,-560,160,18);x.lineWidth=5;for(let i=0;i<14;i++){const y=-20-i*48,w=250-i*16;x.beginPath();x.moveTo(-w,y);x.lineTo(w*0.8,y-48);x.moveTo(w,y);x.lineTo(-w*0.8,y-48);x.stroke();}x.fillRect(-6,-1000,12,100);x.restore();}
function paris(t){x.fillStyle=grad(0,H,['#9cc3e0','#f1dcc0']);x.fillRect(0,0,W,H);city(1060,'#b8a48c',false);x.fillStyle='#8a7a66';x.fillRect(0,1060,W,H-1060);}
function antenna(cx,cy,t){x.strokeStyle='#ffd23f';x.lineWidth=7;for(let k=0;k<3;k++){const ph=(t*1.2+k/3)%1;x.globalAlpha=1-ph;x.beginPath();x.arc(cx,cy,40+ph*260,-2.6,-0.5);x.stroke();}x.globalAlpha=1;}
